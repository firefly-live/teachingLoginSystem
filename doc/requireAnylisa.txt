void Teacher::scoreGrades(){
    print("\n课程\t课程id号\n");
    for(auto cour: courses)
    {
        print("{}\n",cour->showInformation());//展示当前课程
        cour->showStudent();//展示该课程有哪些学生
        cour->scoreGrades();//进入course方法打分
    }
}//给学生打分，传入course
